import pandas as pd
from gensim.models import Word2Vec
from sklearn.decomposition import PCA
from sklearn.metrics.pairwise import cosine_similarity
import matplotlib.pyplot as plt
import numpy as np

# 1. Load CSV dataset
df = pd.read_csv("dataset.csv")  # CHANGE HERE: enter your CSV filename

print("Dataset:")
print(df.head())
print("Available columns:", df.columns.tolist())

# 2. Select the column containing product descriptions/text
# CHANGE HERE: choose the text/description column from your CSV.
# Example: df.columns[0] means the first column. You can instead use the exact CSV header.
TEXT_COLUMN = df.columns[0]

products = df[TEXT_COLUMN].dropna().astype(str).tolist()

# Tokenize descriptions
sentences = [p.lower().split() for p in products if p.strip()]

if not sentences:
    raise ValueError("No non-empty descriptions found in the CSV file.")

# Train Word2Vec
model = Word2Vec(
    sentences,
    vector_size=50,
    window=3,
    min_count=1,
    workers=1
)

# Find similar words
print("\nVocabulary:", list(model.wv.index_to_key))
# CHANGE HERE: replace this search word with a word that occurs in your dataset.
search_word = model.wv.index_to_key[0]
if search_word in model.wv:
    print("Similar words to", search_word, ":")
    print(model.wv.most_similar(search_word, topn=min(3, len(model.wv))))
else:
    print("The search word is not present in this dataset.")

# Product recommendation based on text similarity
def recommend(query):
    query_words = [w for w in query.lower().split() if w in model.wv]

    if not query_words:
        print("No matching words found in the model vocabulary.")
        return

    query_vector = np.mean([model.wv[w] for w in query_words], axis=0)
    scores = []

    for product in products:
        product_words = [w for w in product.lower().split() if w in model.wv]
        if not product_words:
            scores.append(-1)
            continue

        product_vector = np.mean([model.wv[w] for w in product_words], axis=0)
        score = cosine_similarity([query_vector], [product_vector])[0][0]
        scores.append(score)

    best = np.argsort(scores)[::-1][:3]
    print("\nRecommended matching descriptions:")
    for i in best:
        print(products[i])

# CHANGE HERE: enter a short query relevant to the text in your CSV.
recommend("")

# PCA visualization
words = list(model.wv.index_to_key)
vectors = np.array([model.wv[w] for w in words])

if len(words) >= 2:
    points = PCA(n_components=2).fit_transform(vectors)
    plt.figure()
    plt.scatter(points[:, 0], points[:, 1])
    for i, word in enumerate(words):
        plt.annotate(word, points[i])
    plt.title("Word2Vec Visualization using PCA")
    plt.show()
