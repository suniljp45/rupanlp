import pandas as pd
from sklearn.feature_extraction.text import CountVectorizer, TfidfVectorizer
from sklearn.decomposition import LatentDirichletAllocation, TruncatedSVD

# 1. Load CSV dataset
df = pd.read_csv("dataset.csv")  # CHANGE HERE: enter your CSV filename

print("Dataset:")
print(df.head())
print("Available columns:", df.columns.tolist())

# 2. Select the column containing text/documents
# CHANGE HERE: choose the text column from your CSV.
# Example: df.columns[0] means the first column. You can instead write "your exact CSV header".
TEXT_COLUMN = df.columns[0]

documents = df[TEXT_COLUMN].dropna().astype(str).tolist()

if len(documents) < 2:
    raise ValueError("At least two non-empty text documents are required.")

# 3. LDA Topic Modeling (uses Count / Bag of Words)
cv = CountVectorizer(stop_words="english")
X = cv.fit_transform(documents)

if X.shape[1] < 2:
    raise ValueError("Not enough distinct words for topic modelling.")

n_topics = 2
lda = LatentDirichletAllocation(n_components=n_topics, random_state=42)
lda.fit(X)

words = cv.get_feature_names_out()

print("\nLDA Topics:")
for i, topic in enumerate(lda.components_):
    top_words = [words[j] for j in topic.argsort()[-5:][::-1]]
    print("Topic", i + 1, ":", top_words)

# 4. LSA Topic Modeling (uses TF-IDF and SVD)
tv = TfidfVectorizer(stop_words="english")
X2 = tv.fit_transform(documents)

lsa_topics = min(n_topics, X2.shape[0] - 1, X2.shape[1] - 1)
if lsa_topics < 1:
    raise ValueError("Not enough rows or unique words for LSA.")

lsa = TruncatedSVD(n_components=lsa_topics, random_state=42)
lsa.fit(X2)

words = tv.get_feature_names_out()

print("\nLSA Topics:")
for i, topic in enumerate(lsa.components_):
    top_words = [words[j] for j in topic.argsort()[-5:][::-1]]
    print("Topic", i + 1, ":", top_words)
