import pandas as pd
import time
from sklearn.model_selection import train_test_split
from sklearn.feature_extraction.text import CountVectorizer
from sklearn.naive_bayes import MultinomialNB
from sklearn.pipeline import make_pipeline
from sklearn.metrics import accuracy_score, confusion_matrix

# 1. Load CSV dataset
df = pd.read_csv("dataset.csv")  # CHANGE HERE: enter your CSV filename

print("Dataset:")
print(df.head())
print("Available columns:", df.columns.tolist())

# 2. Select the news/headline text column and category column
# CHANGE HERE: select the column containing news text/headlines.
# Example: df.columns[0] means the first column. You can instead use the exact CSV header.
TEXT_COLUMN = df.columns[0]

# CHANGE HERE: select the column containing news categories/classes.
LABEL_COLUMN = df.columns[1]

df = df.dropna(subset=[TEXT_COLUMN, LABEL_COLUMN]).copy()
texts = df[TEXT_COLUMN].astype(str).tolist()
labels = df[LABEL_COLUMN].astype(str).tolist()

# 3. Split dataset
X_train, X_test, y_train, y_test = train_test_split(
    texts, labels, test_size=0.25, random_state=42
)

# 4. Train Naive Bayes model
model = make_pipeline(CountVectorizer(), MultinomialNB())
model.fit(X_train, y_train)

# 5. Evaluate model
pred = model.predict(X_test)
print("\nAccuracy:", accuracy_score(y_test, pred))
print("Confusion Matrix:\n", confusion_matrix(y_test, pred))

# 6. Predict categories for new headlines
headlines = [
    "Global stock markets rally after decision",
    "National team wins football final",
    "Space rocket launches successfully",
    "Diplomatic summit agreement signed"
]

start = time.time()
predictions = model.predict(headlines)
elapsed_ms = (time.time() - start) * 1000

print("\nNEWS CATEGORY PREDICTIONS")
for headline, label in zip(headlines, predictions):
    print(headline, "->", label)

print("Total inference time (ms):", round(elapsed_ms, 2))
